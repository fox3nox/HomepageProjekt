function addObjectToColumns(index, object, tileSize, bucketName) {
  const first = Math.max(0, Math.floor(object.x / tileSize));
  const last = Math.min(
    index.length - 1,
    Math.floor((object.x + object.width - 0.0001) / tileSize),
  );

  for (let col = first; col <= last; col += 1) {
    index[col][bucketName].push(object);
  }
