      entryVelocityY: portal.entryVelocityY,
    });
    return {
      id: portal.id,
      type: "mode",
      mode: player.mode,
    };
  }

  if (portal.type === "speed") {
    setPlayerSpeed(player, portal.speed);
    return {
      id: portal.id,
      type: "speed",
      speed: player.velocityX,
    };
  }

  throw new Error(`Unknown portal type '${portal.type}'`);
}

function processPortals(player, portals) {
  const events = [];
  const playerBox = getPlayerHitbox(player);

  for (const portal of portals ?? []) {
    if (player.triggeredPortalIds.has(portal.id)) continue;
    if (!overlapsAABB(playerBox, portal)) continue;

    player.triggeredPortalIds.add(portal.id);
    events.push(applyPortal(player, portal));
  }

  return events;
}

/**
 * One deterministic 120-Hz physics step.
 * `nearbyWorld` is a collision slice from level-loader.js.
 */
function stepPhysics(player, nearbyWorld, dt, killBounds = {}) {
  const solids = nearbyWorld.solids ?? [];
  const spikes = nearbyWorld.spikes ?? [];
  const portals = nearbyWorld.portals ?? [];

  const killTop = Number.isFinite(killBounds.killTop) ? killBounds.killTop : -Infinity;
  const killBottom = Number.isFinite(killBounds.killBottom)
    ? killBounds.killBottom
    : Number.isFinite(killBounds)
      ? killBounds
      : Infinity;

  player.landedThisStep = false;
  player.jumpedThisStep = false;
  player.modeChangedThisStep = false;
  player.speedChangedThisStep = false;
  player.pulseCooldownRemaining = Math.max(0, player.pulseCooldownRemaining - dt);

  carryByGroundMotion(player, solids);

  const wasOnGround = player.onGround;
  const previousGroundId = player.groundId;
  const previousX = player.x;
  const previousY = player.y;
  const previousRight = previousX + player.width;

  applyModeInput(player);
  integrateVerticalVelocity(player, dt);

  player.x += player.velocityX * dt;
  player.y += player.velocityY * dt;

  const landing = findLandingSurface(player, solids, previousY);
  resolveLanding(player, landing, wasOnGround, previousGroundId);

  for (const solid of solids) {
    if (collidesFatallyWithSolid(player, solid)) {
      player.actionPressed = false;
      return death("solid", solid.id);
    }
  }

  // Swept left-face check protects against future high-speed tunnelling.
  if (player.velocityX > 0) {
    const newRight = player.x + player.width;
    for (const solid of solids) {
      if (!solid.lethalSides) continue;
      const crossedLeftFace = previousRight <= solid.x && newRight > solid.x;
      const verticalOverlap =
        player.y < solid.y + solid.height &&
        player.y + player.height > solid.y;
      if (crossedLeftFace && verticalOverlap) {
        player.actionPressed = false;
        return death("solid", solid.id);
      }
    }
  }

  const playerBox = getPlayerHitbox(player);
  for (const spike of spikes) {
    if (overlapsAABB(playerBox, getSpikeHitbox(spike))) {
      player.actionPressed = false;
      return death("spike", spike.id);
    }
  }

  if (player.y < killTop || player.y > killBottom) {
    player.actionPressed = false;
    return death("void", null);
  }

  const portalEvents = processPortals(player, portals);

  if (!player.jumpedThisStep && player.mode === MODES.RUN) {
    player.jumpBufferRemaining = Math.max(0, player.jumpBufferRemaining - dt);
  }

  player.actionPressed = false;

  return {
    died: false,
    cause: null,
    objectId: null,
    portalEvents,
  };
}
