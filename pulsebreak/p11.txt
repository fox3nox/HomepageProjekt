function modeColor(mode) {
  if (mode === MODES.FLIP) return theme.violet;
  if (mode === MODES.GLIDE) return theme.cyan;
  if (mode === MODES.PULSE) return "#ff66ce";
  return theme.magenta;
}


function bestStorageKey(levelDefinition = currentLevelDefinition) {
  return `pulsebreak.best.${levelDefinition.id}`;
}

function readBestProgress(levelDefinition = currentLevelDefinition) {
  try {
    const value = Number(globalThis.localStorage?.getItem(bestStorageKey(levelDefinition)) ?? 0);
    return Number.isFinite(value) ? Math.max(0, Math.min(100, value)) : 0;
  } catch {
    return 0;
  }
}

function writeBestProgress(value, { forcePersist = false } = {}) {
  bestProgress = Math.max(bestProgress, Math.max(0, Math.min(100, value)));
  const wholePercent = Math.floor(bestProgress);

  // Rendering runs at 60–120+ FPS. Persist only when the visible integer
  // best actually improves (or at an explicit lifecycle boundary), rather
  // than hammering synchronous localStorage every frame.
  if (forcePersist || wholePercent > persistedBestWholePercent) {
    try {
      globalThis.localStorage?.setItem(bestStorageKey(), bestProgress.toFixed(2));
      persistedBestWholePercent = Math.max(persistedBestWholePercent, wholePercent);
    } catch {}
  }

  const cardBest = levelBestEls[currentLevelIndex];
  setText(cardBest, `BEST ${wholePercent}%`);
}

function syncLevelPresentation() {
  theme = { ...BASE_THEME, ...(currentLevelDefinition.theme ?? {}) };
  audio.configure({
    bpm: currentLevelDefinition.bpm,
    beatsPerBar: currentLevelDefinition.music?.beatsPerBar ?? 4,
    rootFrequency: currentLevelDefinition.music?.rootFrequency ?? 55,
    intensity: currentLevelDefinition.music?.intensity ?? 2,
  });

  setText(levelNameEl, currentLevelDefinition.name.toUpperCase());
  setText(levelSubtitleEl, currentLevelDefinition.subtitle ?? "");
  setText(levelDifficultyEl, `${currentLevelDefinition.difficultyLabel ?? "LEVEL"} · ${currentLevelDefinition.columns} TILES`);

  levelButtons.forEach((button, index) => {
    if (!button) return;
    const selected = index === currentLevelIndex;
    button.classList?.toggle("selected", selected);
    button.setAttribute?.("aria-pressed", selected ? "true" : "false");
  });

  levelBestEls.forEach((element, index) => {
    if (!element) return;
    setText(element, `BEST ${Math.floor(readBestProgress(LEVELS[index]))}%`);
  });
}

function selectLevel(index) {
  if (!Number.isInteger(index) || index < 0 || index >= LEVELS.length) return;
  if (gameState !== GAME_STATES.MENU) return;

  currentLevelIndex = index;
  currentLevelDefinition = LEVELS[index];
  bestProgress = readBestProgress(currentLevelDefinition);
  runStartBestProgress = bestProgress;
  persistedBestWholePercent = Math.floor(bestProgress);
  syncLevelPresentation();
  resetRun({ keepStats: false, clearEffects: true });
  updateInterface();
}

function setText(element, value) {
  if (!element) return;
  const text = String(value);
  if (element.textContent !== text) element.textContent = text;
}

function setStyleWidth(element, value) {
  if (!element?.style) return;
  if (element.style.width !== value) element.style.width = value;
}

function setHidden(element, hidden) {
  if (!element) return;
  const currentlyHidden = element.classList?.contains?.("hidden") ?? false;
  if (currentlyHidden !== hidden) element.classList?.toggle("hidden", hidden);
  const ariaValue = hidden ? "true" : "false";
  if (element.getAttribute?.("aria-hidden") !== ariaValue) element.setAttribute?.("aria-hidden", ariaValue);
}

function currentProgress() {
  if (!world || !player) return 0;
  if (gameState === GAME_STATES.COMPLETE) return 100;
  return progressPercent(player.x, world.spawn.x, world.finishX);
}

function resetDirectorState(progress = 0) {
  const safeProgress = Math.max(0, Math.min(100, Number(progress) || 0));
  reachedMilestones.clear();
  for (const threshold of MILESTONE_THRESHOLDS) {
    if (safeProgress >= threshold) reachedMilestones.add(threshold);
  }
  lastDirectorProgress = safeProgress;
  currentDirectorSection = directorSection(safeProgress);
  milestoneFlash = 0;
  milestoneTimer = 0;
  milestoneTitle = "";
  milestoneSubtitle = "";
  finishing = false;
  finishSequenceTimer = 0;
  pendingCompletionGrade = "";
  pendingCompletionStats = "";
  audio.setDrive?.(directorDrive(safeProgress));
  const baseIntensity = currentLevelDefinition.music?.intensity ?? 2;
  audio.setIntensity?.(Math.min(3, baseIntensity + (currentDirectorSection >= 2 ? 1 : 0)));
}
