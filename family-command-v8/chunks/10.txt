        const r=img[k]/255, g=img[k+1]/255, b=img[k+2]/255;
        const max=Math.max(r,g,b), min=Math.min(r,g,b);
        const sat=max>0?(max-min)/max:0;
        const lum=(r+g+b)/3;
        if(sat>.30&&lum<.88){
          mask[y*w+x]=1;
          xHist[x]++;
          yHist[y]++;
        }
      }
    }

    const ys=ttLongest(ttSegments(yHist,Math.max(18,Math.floor(w*.055))),Math.floor(h*.025)).sort((a,b)=>a[0]-b[0]);

    let pair=null;
    outer:

        if(gap>3&&gap<h*.08){
          const ratio=(a[1]-a[0]+1)/(b[1]-b[0]+1);
          if(ratio>.9&&ratio<2.6){
            pair=[a,b];
            break outer;
          }
        }
      }
    }
    if(!pair)return{detected:false,reason:'Morgen-/Nachmittagsraster nicht sicher gefunden'};

    const [morning,afternoon]=pair;

      for(let i=0;i<n;i++){

          seg[0]+(seg[1]+1-seg[0])*i/n,
          seg[0]+(seg[1]+1-seg[0])*(i+1)/n
        ]);
      }
    }

    const matrix=[], fractions=[];
